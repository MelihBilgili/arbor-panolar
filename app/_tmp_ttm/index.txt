// Supabase Edge Function: teklif-talep-mail  (2026-09-09)
// Teklif Talepleri formundaki üç düğme çağırır. Gövde: { tip, blok }
//   blok="YD2" (varsayılan · AK + MKB):
//     tip="talep"           → AK "Teslim Tarihi İste"  → MKB'ye mail, termin = +3 çalışma saati
//     tip="oncelik_sor"     → MKB "Öncelik İste"       → MB'ye mail
//     tip="oncelik_bildir"  → MB "Öncelik Kaydet"      → MKB'ye mail, termin = +1 çalışma saati
//   blok="YD1" (MN = Mustafa Nas · AK ve MKB rollerinin ikisi de onda):
//     tip="talep"           → MAİL YOK (gönderen ile alıcı aynı kişi olurdu) → {ok:true, atlandi:true}
//     tip="oncelik_sor"     → MB'ye mail (aynı)
//     tip="oncelik_bildir"  → MN'ye mail, termin = +1 çalışma saati
// Termin, referans panosundaki `parametreler.calisma_saatleri` / `calisma_gunleri`
// değerlerine göre hesaplanır (Türkiye saati, UTC+3).
// Secret: RESEND_API_KEY (paylaşımlı), opsiyonel MAIL_FROM / MAIL_CC.

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";

/* ARBOR-MAIL-HAT (2026-08-31): mail Resend yerine arbor-mail edge fn uzerinden gider;
   arbor-mail → Zapier catch hook → Outlook "Send Email" (int@arbor.com.tr kutusu).
   Gonderen artik gercekten int@arbor.com.tr; dogrulanmamis alan sorunu yok. */
const MAIL_URL = `${Deno.env.get("SUPABASE_URL")}/functions/v1/arbor-mail`;
const SB_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") || "";

const RESEND_API_KEY = Deno.env.get("RESEND_API_KEY");
const FROM = Deno.env.get("MAIL_FROM") || "Arbor Panolar <onboarding@resend.dev>";
const CC   = Deno.env.get("MAIL_CC")   || "int@arbor.com.tr";
const SB_URL = Deno.env.get("SUPABASE_URL") || "";

const MB_MAIL  = "int@arbor.com.tr";
const MKB_MAIL = "melike.basaran@arbor.com.tr";   // Yurt Dışı 2 teklif sorumlusu
const MN_MAIL  = "mustafa.nas@arbor.com.tr";      // Yurt Dışı 1 sorumlusu

function ARB_OK(req: Request): boolean {
  const h = req.headers.get("authorization") || "";
  const t = h.replace(/^Bearer\s+/i, "");
  const p = t.split(".");
  if (p.length !== 3) return false;
  try {
    const j = JSON.parse(atob(p[1].replace(/-/g, "+").replace(/_/g, "/")));
    if (j.role === "service_role") return true;
    return typeof j.email === "string" && j.email.toLowerCase().endsWith("@arbor.com.tr");
  } catch (_e) { return false; }
}

/* ---------- çalışma takvimi ---------- */
const GUN_NO: Record<string, number> = { "Paz": 0, "Pzt": 1, "Sal": 2, "Çar": 3, "Car": 3, "Per": 4, "Cum": 5, "Cmt": 6 };
const TR_OFFSET_MIN = 180; // Europe/Istanbul, kalıcı UTC+3

type Aralik = { bas: number; bit: number }; // gün içi dakika

function saatleriCoz(s: string): Aralik[] {
  // "08:00 - 12:00; 13:00 - 18:00"
  const out: Aralik[] = [];
  for (const par of String(s || "").split(";")) {
    const m = par.match(/(\d{1,2}):(\d{2})\s*-\s*(\d{1,2}):(\d{2})/);
    if (m) out.push({ bas: (+m[1]) * 60 + (+m[2]), bit: (+m[3]) * 60 + (+m[4]) });
  }
  if (!out.length) out.push({ bas: 8 * 60, bit: 12 * 60 }, { bas: 13 * 60, bit: 18 * 60 });
  out.sort((a, b) => a.bas - b.bas);
  return out;
}

function gunleriCoz(v: unknown): Set<number> {
  const s = new Set<number>();
  const arr = Array.isArray(v) ? v : ["Pzt", "Sal", "Çar", "Per", "Cum"];
  for (const g of arr) {
    const k = String(g).trim();
    if (k in GUN_NO) s.add(GUN_NO[k]);
    else if (/^ça?r/i.test(k)) s.add(3);
  }
  if (!s.size) [1, 2, 3, 4, 5].forEach((d) => s.add(d));
  return s;
}

/** UTC ms → TR "yerel" ms (aritmetik kolaylığı için kaydırılmış) */
const toTR = (ms: number) => ms + TR_OFFSET_MIN * 60000;
const toUTC = (ms: number) => ms - TR_OFFSET_MIN * 60000;

function calismaSaatiEkle(baslangicUTC: number, saat: number, aralik: Aralik[], gunler: Set<number>): number {
  let kalan = Math.round(saat * 60); // dakika
  let t = toTR(baslangicUTC);
  let guard = 0;
  while (kalan > 0 && guard++ < 2000) {
    const d = new Date(t);
    const gun = d.getUTCDay();
    const dk = d.getUTCHours() * 60 + d.getUTCMinutes();
    const gunBasi = t - dk * 60000;
    if (!gunler.has(gun)) { t = gunBasi + 24 * 60 * 60000 + aralik[0].bas * 60000; continue; }
    let yerlesti = false;
    for (const a of aralik) {
      if (dk < a.bas) { t = gunBasi + a.bas * 60000; yerlesti = true; break; }
      if (dk < a.bit) {
        const musait = a.bit - dk;
        if (kalan <= musait) { t = gunBasi + (dk + kalan) * 60000; kalan = 0; }
        else { kalan -= musait; t = gunBasi + a.bit * 60000; }
        yerlesti = true;
        break;
      }
    }
    if (!yerlesti) t = gunBasi + 24 * 60 * 60000 + aralik[0].bas * 60000; // gün bitti → ertesi gün
  }
  return toUTC(t);
}

function trFmt(utcMs: number): string {
  const d = new Date(toTR(utcMs));
  const p = (n: number) => String(n).padStart(2, "0");
  return `${p(d.getUTCDate())}.${p(d.getUTCMonth() + 1)}.${d.getUTCFullYear()} ${p(d.getUTCHours())}:${p(d.getUTCMinutes())}`;
}

async function takvimOku(): Promise<{ aralik: Aralik[]; gunler: Set<number> }> {
  try {
    const r = await fetch(`${SB_URL}/rest/v1/parametreler?select=anahtar,deger&anahtar=in.(calisma_saatleri,calisma_gunleri)`, {
      headers: { apikey: SB_KEY, Authorization: `Bearer ${SB_KEY}` },
    });
    const rows = await r.json();
    const m: Record<string, unknown> = {};
    for (const x of rows) m[x.anahtar] = x.deger;
    return { aralik: saatleriCoz(String(m["calisma_saatleri"] ?? "")), gunler: gunleriCoz(m["calisma_gunleri"]) };
  } catch (_e) {
    return { aralik: saatleriCoz(""), gunler: gunleriCoz(null) };
  }
}

serve(async (req) => {
  const cors = {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
    "Access-Control-Allow-Methods": "POST, OPTIONS",
  };
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });
  if (!ARB_OK(req)) return new Response(JSON.stringify({ error: "yetkisiz" }), { status: 401, headers: { ...cors, "Content-Type": "application/json" } });
  const json = (o: unknown, status = 200) =>
    new Response(JSON.stringify(o), { status, headers: { ...cors, "Content-Type": "application/json" } });

  try {
    const { tip, blok } = await req.json();
    if (!SB_KEY) return json({ error: "SUPABASE_SERVICE_ROLE_KEY tanımlı değil" }, 500);

    // Blok: bilinmeyen/eksik değer YD2'ye düşer (eski çağrılarla geriye dönük uyumlu).
    const B = blok === "YD1" ? "YD1" : "YD2";
    const BAD = B === "YD1" ? "Yurt Dışı 1" : "Yurt Dışı 2";
    // Teklif tarafını kim yürütüyor: YD1'de MN (Mustafa Nas), YD2'de MKB (Melike Başaran).
    const TEKLIF_MAIL = B === "YD1" ? MN_MAIL : MKB_MAIL;
    const HITAP = B === "YD1" ? "Mustafa Bey," : "Melike Hanım,";

    // YD1'de AK ve MKB rollerinin ikisi de MN'de → "Teslim Tarihi İste" maili
    // gönderen ile alıcıyı aynı kişi yapardı; mail atılmaz.
    if (tip === "talep" && B === "YD1") return json({ ok: true, atlandi: "YD1 talep maili gönderilmez" });

    let to: string, subject: string, metin: string;

    if (tip === "oncelik_sor") {
      to = MB_MAIL;
      subject = `Teklif Talepleri (${BAD}) — öncelik grubu bekleyen satır`;
      metin = `Teklif Talepleri formunun ${BAD} bloğuna öncelik grubunu seçmen gereken yeni bir satır işlendi.`;
    } else if (tip === "talep" || tip === "oncelik_bildir") {
      const { aralik, gunler } = await takvimOku();
      const saat = tip === "talep" ? 3 : 1;
      const son = trFmt(calismaSaatiEkle(Date.now(), saat, aralik, gunler));
      to = TEKLIF_MAIL;
      if (tip === "talep") {
        subject = `Teklif Talepleri (${BAD}) — yeni proje eklendi`;
        metin =
          HITAP + "\n\n" +
          `Teklif Talepleri formunun ${BAD} bloğuna yeni bir proje eklenmiştir. ${son}'e kadar size ait olan bilgileri işleyip ` +
          "Öncelik İste düğmesine basmanızı rica ederim.\n\nMelih Bilgili.";
      } else {
        subject = `Teklif Talepleri (${BAD}) — yeni öncelik grupları işlendi`;
        metin =
          HITAP + "\n\n" +
          `Teklif Talepleri formunun ${BAD} bloğuna yeni öncelik grupları işlenmiştir. ${son}'e kadar size ait olan bilgileri işleyip ` +
          "Teslim Tarihlerini Kaydet düğmesine basmanızı rica ederim.\n\nMelih Bilgili.";
      }
    } else {
      return json({ error: "bilinmeyen tip" }, 400);
    }

    const html = "<div style=\"font-family:'Segoe UI',system-ui,sans-serif;font-size:14px;line-height:1.55\">" +
      metin.split("\n").map((l) => (l.trim() === "" ? "<br>" : `<div>${l.replace(/&/g, "&amp;").replace(/</g, "&lt;")}</div>`)).join("") +
      "</div>";

    const payload: Record<string, unknown> = { to: [to], subject, text: metin, html };
    if (to !== CC) payload.cc = [CC];
    const r = await fetch(MAIL_URL, {
      method: "POST",
      headers: { Authorization: `Bearer ${SB_KEY}`, "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });
    const body = await r.text();
    if (!r.ok) return json({ error: `Mail ${r.status}: ${body}` }, 502);
    return json({ ok: true, to, subject });
  } catch (e) {
    return json({ error: String(e) }, 500);
  }
});
